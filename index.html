#!/usr/bin/env python3 """ Vehicle State Estimator with Sensor Fusion IMU (BNO085) + GNSS (UM981) → Reliable Position & Heading APPROACH: - Position: GNSS primary, IMU dead-reckoning backup - Heading: GNSS course when moving, IMU magnetometer when stationary - Complementary filter for smooth fusion - Outlier rejection and quality scoring """ import time import json import math import logging import signal import serial import pynmea2 import paho.mqtt.client as mqtt from collections import deque from adafruit_blinka.microcontroller.generic_linux.i2c import I2C as LinuxI2C from adafruit_bno08x.i2c import BNO08X_I2C from adafruit_bno08x import ( BNO_REPORT_ACCELEROMETER, BNO_REPORT_GYROSCOPE, BNO_REPORT_MAGNETOMETER, BNO_REPORT_ROTATION_VECTOR, ) # ============== CONFIG ============== MQTT_HOST = "broker.emqx.io" MQTT_PORT = 1883 MQTT_TOPIC = "robot/fused_state" MQTT_USER = "imu" MQTT_PASS = "emcx_test" IMU_RATE_HZ = 20 PUBLISH_HZ = 10 # GNSS SERIAL_PORT = "/dev/ttyAMA4" SERIAL_BAUD = 115200 MIN_GNSS_QUALITY = 1 # 1=GPS, 4=RTK fixed MIN_SATELLITES = 4 # Heading fusion MOVING_SPEED_THRESHOLD = 1.0 # m/s (trust GNSS course above this) IMU_YAW_OFFSET_DEG = 222.0 # your calibrated mounting offset MAG_TRUST_WEIGHT = 0.3 # when stationary, trust mag at 30% # Position fusion MAX_GNSS_AGE = 5.0 # seconds before falling back to dead reckoning # ===================================== logging.basicConfig(level=logging.INFO, format="%(asctime)s [%(levelname)s] %(message)s") logger = logging.getLogger("fusion") # ============== I2C FIX ============== class LinuxI2CCompat(LinuxI2C): def try_lock(self): return True def unlock(self): pass # ============== MQTT ============== class MQTTPublisher: def __init__(self): self.client = mqtt.Client( client_id="vehicle_fusion_pi", protocol=mqtt.MQTTv311, callback_api_version=mqtt.CallbackAPIVersion.VERSION1 ) self.client.username_pw_set(MQTT_USER, MQTT_PASS) self.client.connect(MQTT_HOST, MQTT_PORT, 30) self.client.loop_start() logger.info("MQTT connected") def publish(self, payload): self.client.publish(MQTT_TOPIC, json.dumps(payload), qos=0) def shutdown(self): self.client.loop_stop() self.client.disconnect() # ============== GNSS ============== class GNSS: def __init__(self): self.ser = serial.Serial(SERIAL_PORT, SERIAL_BAUD, timeout=0.05) self.buffer = bytearray() self.last_fix = None self.last_valid_time = None logger.info("GNSS online") def read_stream(self): data = self.ser.read(256) if not data: return self.buffer.extend(data) while b"\n" in self.buffer: line_bytes, _, self.buffer = self.buffer.partition(b"\n") line = line_bytes.decode("ascii", errors="ignore").strip() if not line.startswith("$") or "*" not in line: continue if not (line.startswith("$GNGGA") or line.startswith("$GPGGA")): continue try: msg = pynmea2.parse(line) except: continue if msg.gps_qual < MIN_GNSS_QUALITY or msg.num_sats < MIN_SATELLITES: continue self.last_fix = { "lat": float(msg.latitude), "lon": float(msg.longitude), "alt": float(msg.altitude), "fix": int(msg.gps_qual), "sats": int(msg.num_sats), "hdop": float(msg.horizontal_dil), } self.last_valid_time = time.time() def get_fix(self): if not self.last_fix: return None age = time.time() - self.last_valid_time if self.last_valid_time else 999 if age > MAX_GNSS_AGE: return None # stale return {**self.last_fix, "age": age} # ============== IMU ============== class IMU: def __init__(self): self.i2c = LinuxI2CCompat(0) self.bno = BNO08X_I2C(self.i2c) # -------- IMU OFFSETS (SUBTRACTED) -------- # Uncomment and calibrate these for your setup: self.accel_offset = [0.0, 0.0, 0.0] # m/s² self.gyro_offset = [0.0, 0.0, 0.0] # rad/s self.mag_offset = [0.0, 0.0, 0.0] # uT (hard-iron) # Example calibrated values (from your first script): # self.accel_offset = [0.24, -0.46, -0.20] # self.gyro_offset = [0.0, 0.0, 0.0] # self.mag_offset = [-9.8, -85.0, -13.0] us = int(1_000_000 / IMU_RATE_HZ) self.bno.enable_feature(BNO_REPORT_ACCELEROMETER, us) self.bno.enable_feature(BNO_REPORT_GYROSCOPE, us) self.bno.enable_feature(BNO_REPORT_MAGNETOMETER, us) self.bno.enable_feature(BNO_REPORT_ROTATION_VECTOR, us) logger.info("IMU online (9-axis)") @staticmethod def quat_to_heading(q): """Convert quaternion to compass heading (0-360°, N=0)""" qx, qy, qz, qw = q yaw = math.atan2(2 * (qw * qz + qx * qy), 1 - 2 * (qy * qy + qz * qz)) compass = (90 - math.degrees(yaw)) % 360 return (compass + IMU_YAW_OFFSET_DEG) % 360 @staticmethod def _apply_offset(vec, offset): """Subtract calibration offset from sensor reading""" if not vec: return vec return tuple(v - o for v, o in zip(vec, offset)) def read(self): quat = self.bno.quaternion accel = self._apply_offset(self.bno.acceleration, self.accel_offset) gyro = self._apply_offset(self.bno.gyro, self.gyro_offset) mag = self._apply_offset(self.bno.magnetic, self.mag_offset) return { "quat": quat, "heading_mag": round(self.quat_to_heading(quat), 2) if quat else None, "accel": accel, "gyro": gyro, "mag": mag, } # ============== VELOCITY ESTIMATOR ============== class VelocityEstimator: """Estimate velocity from position history""" def __init__(self, window=5): self.history = deque(maxlen=window) def update(self, lat, lon, timestamp): self.history.append((lat, lon, timestamp)) def get_speed(self): if len(self.history) < 2: return 0.0 lat1, lon1, t1 = self.history[0] lat2, lon2, t2 = self.history[-1] dt = t2 - t1 if dt < 0.1: return 0.0 # Haversine distance R = 6371000 # Earth radius in meters dlat = math.radians(lat2 - lat1) dlon = math.radians(lon2 - lon1) a = math.sin(dlat/2)**2 + math.cos(math.radians(lat1)) * math.cos(math.radians(lat2)) * math.sin(dlon/2)**2 dist = R * 2 * math.atan2(math.sqrt(a), math.sqrt(1-a)) return dist / dt def get_course(self): """GNSS course over ground (bearing between positions)""" if len(self.history) < 2: return None lat1, lon1, _ = self.history[-2] lat2, lon2, _ = self.history[-1] dlon = math.radians(lon2 - lon1) lat1_r = math.radians(lat1) lat2_r = math.radians(lat2) y = math.sin(dlon) * math.cos(lat2_r) x = math.cos(lat1_r) * math.sin(lat2_r) - math.sin(lat1_r) * math.cos(lat2_r) * math.cos(dlon) bearing = (math.degrees(math.atan2(y, x)) + 360) % 360 return bearing # ============== HEADING FUSION ============== class HeadingFusion: """Fuse IMU magnetometer + GNSS course""" def __init__(self): self.fused_heading = None def update(self, imu_heading, gnss_course, speed): if gnss_course is None and imu_heading is None: return self.fused_heading # Decide weights based on speed if speed > MOVING_SPEED_THRESHOLD: # Trust GNSS course heavily when moving gnss_weight = 0.9 else: # Trust magnetometer more when stationary gnss_weight = 1.0 - MAG_TRUST_WEIGHT # Circular mean (handle 359°/1° wraparound) if gnss_course is not None and imu_heading is not None: gnss_rad = math.radians(gnss_course) imu_rad = math.radians(imu_heading) x = gnss_weight * math.cos(gnss_rad) + (1-gnss_weight) * math.cos(imu_rad) y = gnss_weight * math.sin(gnss_rad) + (1-gnss_weight) * math.sin(imu_rad) fused = (math.degrees(math.atan2(y, x)) + 360) % 360 elif gnss_course is not None: fused = gnss_course elif imu_heading is not None: fused = imu_heading else: return self.fused_heading # Low-pass filter for smoothness if self.fused_heading is None: self.fused_heading = fused else: # Handle wraparound diff = (fused - self.fused_heading + 180) % 360 - 180 self.fused_heading = (self.fused_heading + 0.3 * diff) % 360 return self.fused_heading # ============== MAIN ============== def main(): mqtt_pub = MQTTPublisher() imu = IMU() gnss = GNSS() vel_est = VelocityEstimator() heading_fuser = HeadingFusion() running = True signal.signal(signal.SIGTERM, lambda *_: globals().update(running=False)) publish_period = 1.0 / PUBLISH_HZ next_publish = time.monotonic() seq = 0 logger.info("=== Fusion loop started ===") try: while running: # Read sensors continuously gnss.read_stream() imu_data = imu.read() gnss_fix = gnss.get_fix() now = time.monotonic() # Update velocity estimation if gnss_fix: vel_est.update(gnss_fix["lat"], gnss_fix["lon"], time.time()) speed = vel_est.get_speed() gnss_course = vel_est.get_course() if speed > 0.5 else None # Fuse heading fused_heading = heading_fuser.update( imu_data["heading_mag"], gnss_course, speed ) # Position source logic if gnss_fix: pos_source = "gnss" pos_quality = "high" if gnss_fix["fix"] >= 4 else "medium" position = { "lat": gnss_fix["lat"], "lon": gnss_fix["lon"], "alt": gnss_fix["alt"], } else: pos_source = "dead_reckoning" # Could implement DR here pos_quality = "low" position = None # Heading source logic if speed > MOVING_SPEED_THRESHOLD and gnss_course is not None: heading_source = "gnss_course" heading_quality = "high" elif imu_data["heading_mag"] is not None: heading_source = "imu_mag" heading_quality = "medium" else: heading_source = "none" heading_quality = "low" # Publish at fixed rate if now >= next_publish: payload = { "timestamp": int(time.time()), "seq": seq, # FUSED OUTPUTS "position": position, "heading_deg": round(fused_heading, 2) if fused_heading else None, "speed_ms": round(speed, 2), # Quality indicators "pos_source": pos_source, "pos_quality": pos_quality, "heading_source": heading_source, "heading_quality": heading_quality, # Raw sensor data "gnss": gnss_fix, "imu": { "heading_mag": imu_data["heading_mag"], "accel": list(imu_data["accel"]) if imu_data["accel"] else None, "gyro": list(imu_data["gyro"]) if imu_data["gyro"] else None, }, "gnss_course": round(gnss_course, 2) if gnss_course else None, } mqtt_pub.publish(payload) print(json.dumps({ "pos": pos_source, "heading": heading_source, "speed": round(speed, 1), "hdg": round(fused_heading, 1) if fused_heading else None }, indent=2)) seq = (seq + 1) & 0xFFFF next_publish += publish_period time.sleep(0.001) except KeyboardInterrupt: logger.info("Ctrl+C received") finally: mqtt_pub.shutdown() logger.info("Shutdown complete") if __name__ == "__main__": main()
